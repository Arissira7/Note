# 1. 业务分析

在企业的内部，一般都会有很多公司文件，很多时候我们都需要去查询这些文件，然后基于这些文件信息来回答一些问题或者是处理问题。比如公司的规章制度，奖惩制度等等。这里就需要用到AI知识库了。简单来说有了AI助手之后，我们就不需要再人为的去查找某些对应的文件资料了。我们只需要将这些所有的知识库文件都喂给大模型。大模型会将这些文件做向量化存储，然后根据用户的提问直接去查找相关联的文件信息，然后再结合大模型的分析能力最后直接给我们生成想要的信息

![](<images/9. 构建知识库-1.png>)

如上图所示，整个流程中最难的地方就在于更新知识库，需要先上传文件，然后由程序把文件加载，并读取文件内容，再去更新知识库信息 。&#x20;

在实现过程中，用户首先把文件发送给AI Chat聊天助手 ，下一步就通知AI Chat，根据上传的文件加载最新的知识库信息 。这里的核心难点是有两次会话，第一次会话属于上传文件，第二次会话才是去指明要做的事情&#x20;

![](<images/9. 构建知识库-2.png>)

实现上来说，我们在上传文件的时候，先把文件存储在系统中， 存储完之后，由程序读取到文件的信息，比如 文件大小，文件名，文件后缀，文件目录地址等信息，然后记录好之后， 再把信息写入到这个用户对应的AI助手记忆机制中。那么在下一次对话更新知识库的时候，就可以根据记忆机制找到我们要上传的文件&#x20;

![](<images/9. 构建知识库-3.png>)

// NewChat 创建聊天业务逻辑实例
func NewChat(svc *svc.ServiceContext) Chat {
    // 初始化处理器列表，包含待办事项处理器
    handlers := []router.Handler{
       chatinternal.NewTodoHandle(svc), // 待办事项处理器
    }
    // 创建多会话内存管理器，支持对话历史记忆
    memory := memoryx.NewMemoryx(func() schema.Memory {
       return memoryx.NewSummaryBuffer(svc.LLMs, 50, memoryx.WithCallback(svc.Callbacks),
          memoryx.WithOutParser(memoryOutput))
    })
    return &chat{
       memory: memory,
       svc:    svc,
       router: router.NewRouter(svc.LLMs, handlers,
          router.WithMemory(memory),          // 添加记忆组件支持
          router.Withcallback(svc.Callbacks), // 添加回调处理器
          router.WithEmptyHandler(chatinternal.NewDefaultHandler(svc)),
       ), // 添加内存组件支持
    }
}

* 在agent中添加记忆前缀，在/internal/logic/chatinernal/toolx下新增一个basechat.go文件，加入提示词前缀

````go
package chatinternal

import (
    "AIWorkHelper/internal/svc"
    "AIWorkHelper/pkg/langchain"
    "context"
    "github.com/tmc/langchaingo/agents"
    "github.com/tmc/langchaingo/chains"
    "github.com/tmc/langchaingo/tools"
    "strings"
)

type baseChat struct {
    agentsChain chains.Chain
}

func NewBaseChat(svc *svc.ServiceContext, tools []tools.Tool) *baseChat {
    return &baseChat{
       agentsChain: agents.NewExecutor(agents.NewOneShotAgent(svc.LLMs, tools, agents.WithPromptPrefix(_defaultMrklPrefix))),
    }
}

func (t *baseChat) Chains() chains.Chain {
    return chains.NewTransform(t.transform, nil, nil)
}

func (t *baseChat) transform(ctx context.Context, inputs map[string]any,
    opts ...chains.ChainCallOption) (map[string]any,
    error) {

    for s, a := range inputs {
       if _, ok := a.(string); !ok {
          delete(inputs, s)
       }
    }

    outPut, err := t.agentsChain.Call(ctx, inputs, opts...)
    if err != nil {
       return nil, err
    }
    v, ok := outPut["output"]
    if !ok {
       return outPut, nil
    }

    text := v.(string)

    withoutJSONStart := strings.Split(text, "```json")
    if !(len(withoutJSONStart) > 1) {
       return map[string]any{
          langchain.OutPut: v,
       }, err
    }

    withoutJSONEnd := strings.Split(withoutJSONStart[1], "```")
    if len(withoutJSONEnd) < 1 {
       return map[string]any{
          langchain.OutPut: v,
       }, err
    }

    return map[string]any{
       langchain.OutPut: withoutJSONEnd[0],
    }, nil
}
````

package toolx

import (
    "AIWorkHelper/internal/svc"
    "context"
    "github.com/tmc/langchaingo/callbacks"
    "github.com/tmc/langchaingo/chains"
    "github.com/tmc/langchaingo/vectorstores"
    "github.com/tmc/langchaingo/vectorstores/redisvector"
)

type KnowledgeRetrievalQA struct {
    svc      *svc.ServiceContext
    Callback callbacks.Handler
    store    *redisvector.Store
    qa       chains.Chain
}

func NewKnowledgeRetrievalQA(svc *svc.ServiceContext) *KnowledgeRetrievalQA {
    return &KnowledgeRetrievalQA{svc: svc}
}

func (k *KnowledgeRetrievalQA) Name() string {
    return "knowledge_retrieval_qa"
}

func (k *KnowledgeRetrievalQA) Description() string {
    return `a knowledge retrieval interface.
use it when you need to inquire about work-related policies, such as employee manuals, attendance rules, etc.
keep Chinese output.`
}

func (k *KnowledgeRetrievalQA) Call(ctx context.Context, input string) (string, error) {
    var err error
    if k.qa == nil {
       k.store, err = getKnowledgeStore(ctx, k.svc)
       if err != nil {
          return "", err
       }

       k.qa = chains.NewRetrievalQAFromLLM(k.svc.LLMs, vectorstores.ToRetriever(k.store, 1))
    }

    res, err := chains.Predict(ctx, k.qa, map[string]any{
       "query": input,
    })
    if err != nil {
       return "", err
    }

    return `The following are the consultation results. When outputting, please output the results directly, do not make summaries, keep them in Chinese, and only do the original output:
\n` + res, nil
}
```
